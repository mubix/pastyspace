Also speaks fluent curl.

# random three-word URL
echo "hello" | curl --data-binary @- {{ request.host_url }}
# claim a specific URL
cat notes.txt | curl --data-binary @- {{ request.host_url }}my-notes
# files and images (repeat -F for more), optional text and syntax
curl -F "file=@shot.png" -F "body=optional text" -F "syntax=bash" {{ request.host_url }}
# raw binary upload
curl --data-binary @report.pdf -H "Content-Type: application/pdf" "{{ request.host_url }}?filename=report.pdf"
# read it back
curl {{ request.host_url }}raw/my-notes
